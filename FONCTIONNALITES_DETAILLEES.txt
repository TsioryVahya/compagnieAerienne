   │ reservation_place       │ OneToMany    │ JOIN sur place BETWEEN     │
   │                         │              │ place_debut AND place_fin  │
   │ reservation             │ ManyToOne    │ reservation_place.id_res   │
   │ vol_programmation       │ ManyToOne    │ reservation.id_vol_prog    │

REQUÊTE SQL POUR CA PAR CLASSE :
────────────────────────────────
SELECT 
    c.nom as classe,
    COUNT(DISTINCT rp.id_reservation) as nb_reservations,
    COUNT(rp.place) as places_vendues,
    tv.tarif,
    SUM(tv.tarif) as ca_classe
FROM classe c
INNER JOIN classe_place cp ON c.id = cp.id_classe
INNER JOIN tarif_vol tv ON c.id = tv.id_classe
LEFT JOIN reservation r ON r.id_vol_programmation = tv.id_vol_programmation
LEFT JOIN reservation_place rp ON r.id = rp.id_reservation
    AND rp.place BETWEEN cp.place_debut AND cp.place_fin
WHERE tv.id_vol_programmation = ?
GROUP BY c.id, c.nom, tv.tarif;

REQUÊTE SQL POUR CA PUBLICITÉS D'UN VOL :
─────────────────────────────────────────
SELECT 
    s.nom as societe,
    d.nom as diffusion,
    dp.nombre_diffusions,
    tp.montant as tarif_unitaire,
    (dp.nombre_diffusions * tp.montant) as montant_total
FROM diffusion_programmation dp
INNER JOIN diffusion d ON dp.id_diffusion = d.id
INNER JOIN societe s ON d.id_societe = s.id
INNER JOIN (
    SELECT montant 
    FROM tarif_pub 
    WHERE date_application <= dp.date_programmation
    ORDER BY date_application DESC 
    LIMIT 1
) tp
WHERE dp.id_vol_programmation = ?;

   SELECT 
       vp.id as vol_programmation_id,
       -- CA Réservations
       COALESCE(SUM(tv.tarif), 0) as ca_reservations,
       -- CA Produits Extras
       COALESCE(SUM(vpd.quantite * vpd.prix_unitaire), 0) as ca_produits,
       -- CA Publicités (montant à facturer)
       COALESCE(SUM(dp.nombre_diffusions * tp.montant), 0) as ca_publicites,
       -- Total CA
       (ca_reservations + ca_produits + ca_publicites) as ca_total,
       -- Montants payés

       -- Reste à payer

REQUÊTE SQL COMPLÈTE POUR CA + PAIEMENTS :
──────────────────────────────────────────
SELECT 
    vp.id,
    vp.date_heure,
    
    -- 1. CA Réservations
    COALESCE(SUM(DISTINCT tv.tarif * r.nombre_places), 0) as ca_reservations,
    
    -- 2. CA Produits Extras
    COALESCE(SUM(DISTINCT vpd.quantite * vpd.prix_unitaire), 0) as ca_produits,
    
    -- 3. CA Publicités (à facturer)
    COALESCE(SUM(DISTINCT dp.nombre_diffusions * tp.montant), 0) as ca_publicites,
    
    -- 4. Total CA
    (ca_reservations + ca_produits + ca_publicites) as ca_total,
    
    -- 5. Montants payés
    COALESCE(SUM(DISTINCT pdd.montant), 0) as montant_paye,
    
    -- 6. Reste à payer
    (ca_publicites - montant_paye) as montant_reste,
    
    -- 7. Statut paiement
    CASE 
        WHEN montant_paye = 0 THEN 'Non payé'
        WHEN montant_paye < ca_publicites THEN 'Partiellement payé'
        WHEN montant_paye >= ca_publicites THEN 'Payé intégralement'
    END as statut_paiement

FROM vol_programmation vp

-- Réservations
LEFT JOIN reservation r ON vp.id = r.id_vol_programmation
LEFT JOIN reservation_place rp ON r.id = rp.id_reservation
LEFT JOIN tarif_vol tv ON vp.id = tv.id_vol_programmation 
    AND rp.id_type_passager = tv.id_type_passager

-- Produits extras
LEFT JOIN vente_produit vp2 ON r.id_client = vp2.id_client
LEFT JOIN vente_produit_detail vpd ON vp2.id = vpd.id_vente_produit

-- Publicités
LEFT JOIN diffusion_programmation dp ON vp.id = dp.id_vol_programmation
LEFT JOIN diffusion d ON dp.id_diffusion = d.id
LEFT JOIN (
    SELECT montant FROM tarif_pub 
    WHERE date_application <= CURRENT_DATE 
    ORDER BY date_application DESC LIMIT 1
) tp

-- Paiements
LEFT JOIN payment_diffusion pd ON dp.id = pd.id_diffusion_programmation
LEFT JOIN payment_details_diffusion pdd ON pd.id = pdd.id_payment

WHERE vp.id = ?
GROUP BY vp.id, vp.date_heure;

REQUÊTE SQL POUR CA PRODUITS EXTRAS D'UN VOL :
──────────────────────────────────────────────
SELECT 
    pe.nom as produit,
    SUM(vpd.quantite) as quantite_totale,
    vpd.prix_unitaire,
    SUM(vpd.quantite * vpd.prix_unitaire) as ca_produit
FROM vol_programmation vp
INNER JOIN reservation r ON vp.id = r.id_vol_programmation
INNER JOIN client c ON r.id_client = c.id
INNER JOIN vente_produit vp2 ON c.id = vp2.id_client
INNER JOIN vente_produit_detail vpd ON vp2.id = vpd.id_vente_produit
INNER JOIN produit_extra pe ON vpd.id_produit_extra = pe.id
WHERE vp.id = ?
  AND vp2.date_vente BETWEEN vp.date_heure - INTERVAL '1 day' 
                         AND vp.date_heure + INTERVAL '1 day'
GROUP BY pe.id, pe.nom, vpd.prix_unitaire
ORDER BY ca_produit DESC;

REQUÊTE SQL POUR RAPPORT CA GLOBAL :
────────────────────────────────────
-- Rapport pour un mois donné (ex: Février 2026)
SELECT 
    -- Totaux globaux du mois
    SUM(ca_reservations) as total_ca_reservations,
    SUM(ca_publicites) as total_ca_publicites,
    SUM(pub_paye) as total_pub_paye,
    SUM(pub_reste) as total_pub_reste,
    COUNT(DISTINCT vp.id) as nombre_vols,
    
    -- CA produits extras (requête séparée)
    (SELECT SUM(vpd.quantite * vpd.prix_unitaire)
     FROM vente_produit vp2
     INNER JOIN vente_produit_detail vpd ON vp2.id = vpd.id_vente_produit
     WHERE YEAR(vp2.date_vente) = ? AND MONTH(vp2.date_vente) = ?
    ) as ca_produits_extras,
    
    -- CA total général
    (total_ca_reservations + total_ca_publicites + ca_produits_extras) as ca_total_mois

FROM (
    -- Sous-requête pour chaque vol
    SELECT 
        vp.id,
        vp.date_heure,
        
        -- CA réservations du vol
        COALESCE(SUM(tv.tarif), 0) as ca_reservations,
        
        -- CA publicités du vol
        COALESCE(SUM(dp.nombre_diffusions * tp.montant), 0) as ca_publicites,
        
        -- Montant payé
        COALESCE(SUM(pdd.montant), 0) as pub_paye,
        
        -- Reste à payer
        (ca_publicites - pub_paye) as pub_reste
        
    FROM vol_programmation vp
    LEFT JOIN reservation r ON vp.id = r.id_vol_programmation
    LEFT JOIN reservation_place rp ON r.id = rp.id_reservation
    LEFT JOIN tarif_vol tv ON vp.id = tv.id_vol_programmation 
        AND rp.id_type_passager = tv.id_type_passager
    LEFT JOIN diffusion_programmation dp ON vp.id = dp.id_vol_programmation
    LEFT JOIN (
        SELECT montant FROM tarif_pub 
        WHERE date_application <= CURRENT_DATE 
        ORDER BY date_application DESC LIMIT 1
    ) tp
    LEFT JOIN payment_diffusion pd ON dp.id = pd.id_diffusion_programmation
    LEFT JOIN payment_details_diffusion pdd ON pd.id = pdd.id_payment
    
    WHERE YEAR(vp.date_heure) = ?
      AND MONTH(vp.date_heure) = ?
    
    GROUP BY vp.id, vp.date_heure
) as vol_details;
